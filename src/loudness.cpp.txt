#include <Arduino.h>

// Pino onde o sensor está ligado (fio amarelo no A0)
const int PINO_SENSOR = A0;

// Valor acima do qual consideramos que há "barulho"
// (ajusta este número conforme os teus testes)
const int LIMITE_BARULHO = 100;

int loudness;

void setup()
{
    Serial.begin(9600);  // tem de ser igual ao monitor_speed
}

void loop()
{
    loudness = analogRead(PINO_SENSOR);  // devolve um valor entre 0 e 1023

    Serial.print("Loudness: ");
    Serial.print(loudness);

    if (loudness > LIMITE_BARULHO)
    {
        Serial.println("  -> BARULHO!");
    }
    else
    {
        Serial.println("  -> silêncio");
    }

    delay(200);  // lê 5 vezes por segundo
}
